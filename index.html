async function submitInvestment(projectId) { if (!currentUser) { alert("Vous devez être connecté."); return; } const input = document.getElementById( "investmentAmount-" + projectId ); const paymentSelect = document.getElementById( "paymentMethod-" + projectId ); const message = document.getElementById( "investmentMessage-" + projectId ); const amount = Number(input.value); const method = paymentSelect.value; if (!amount || amount <= 0) { message.innerHTML = `
Entrez un montant valide.
`; return; } if (!method) { message.innerHTML = `
Choisissez un moyen de paiement.
`; return; } /* ========================= CRÉER LA DEMANDE ========================== */ const { data: investment, error: investmentError } = await sb .from("investments") .insert({ user_id: currentUser.id, project_id: projectId, amount: Math.round(amount), status: "pending" }) .select() .single(); if (investmentError) { console.error(investmentError); message.innerHTML = `
❌ ${escapeHtml(investmentError.message)}
`; return; } /* ========================= ENREGISTRER LE PAIEMENT ========================== */ const { error: paymentError } = await sb .from("payments") .insert({ user_id: currentUser.id, investment_id: investment.id, amount: Math.round(amount), method: method, status: "pending" }); if (paymentError) { console.error(paymentError); message.innerHTML = `
❌ Demande créée, mais erreur lors de l'enregistrement du paiement : ${escapeHtml(paymentError.message)}
`; return; } message.innerHTML = `
✅ Votre demande d'investissement a été enregistrée.
💳 Moyen de paiement : ${escapeHtml(method)}
⏳ Paiement en attente de confirmation.
`; input.value = ""; paymentSelect.value = ""; await loadMyInvestments(); }
